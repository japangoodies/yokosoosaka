Import from Facebook

Extract product info from a Facebook post into the admin panel.

How to use:
  1. Open the Facebook post in your browser
  2. Press F12 → click Console tab
  3. Copy the script below, paste it into the Console, press Enter
  4. The data will be copied to your clipboard automatically
  5. Come back here and paste (Ctrl+V) into the field below
  6. Click Parse & Import to Admin
Step 1: Copy this Console Script
// Paste this into Facebook's Console (F12) on a product post function collectFacebookPost() { var data = { url: location.href, images: [] }; // Collect all visible fbcdn images (carousel/main image) var imgs = document.querySelectorAll('img[src*="fbcdn"], img[src*="scontent"]'); imgs.forEach(function(img) { if (img.offsetParent !== null && img.src) { data.images.push(img.src); } }); // Try to find the post text var article = document.querySelector('[role="article"]'); if (article) data.text = article.textContent.trim(); var str = JSON.stringify(data, null, 2); copyToClipboard(str); return str; } function copyToClipboard(str) { var ta = document.createElement('textarea'); ta.value = str; document.body.appendChild(ta); ta.select(); try { document.execCommand('copy'); } catch (e) { prompt('Copy failed. Copy this manually:', str); } document.body.removeChild(ta); console.log('Copied!'); } console.log(collectFacebookPost());

After pasting into Console, the output will be on your clipboard.
